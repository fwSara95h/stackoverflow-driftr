function calcOp(operator) {
  getInputAsNumber();
  var op;
  if (operator == 'add') {
    writeResult(operator1 + operator2);
  } else if (operator == 'sub') {
    writeResult(operator1 - operator2);
  } else if (operator == 'mul') {
    writeResult(operator1 * operator2);
  } else if (operator == 'div') {
    writeResult(operator1 / operator2);
  }
}
